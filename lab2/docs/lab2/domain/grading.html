 10    """
 11    Оценка за конкретный вид контроля (экзамен, зачет, лабораторная).
 12
 13    Attributes:
 14        subject_name (str): Название дисциплины.
 15        score (int): Выставленный балл.
 16        is_exam (bool): Является ли оценка итоговой (экзаменационной).
 17        date_issued (datetime): Точное время выставления оценки (в UTC).
 18    """
 19    MIN_SCORE = 0
 20    MAX_SCORE = 10
 21    MIN_PASSING_SCORE = 4
 22
 23    def __init__(self, subject_name: str, score: int, is_exam: bool = False) -> None:
 24        self.subject_name = subject_name
 25        self.score = score
 26        self.is_exam = is_exam
 27        self.date_issued = datetime.now(timezone.utc)
 28        self._validate_score()
 29
 30    def _validate_score(self) -> None:
 31        """
 32        Внутренний валидатор диапазона оценки.
 33
 34        Raises:
 35            ValueError: Если балл выходит за допустимые границы (0-10).
 36        """
 37        if not (self.MIN_SCORE <= self.score <= self.MAX_SCORE):
 38            raise ValueError(
 39                f"Недопустимый балл: {self.score}. В системе оценок допустимы "
 40                f"значения от {self.MIN_SCORE} до {self.MAX_SCORE}."
 41            )
 42
 43    @property
 44    def is_passed(self) -> bool:
 45        """Бизнес-правило: считается ли предмет сданным (4 балла и выше)."""
 46        return self.score >= self.MIN_PASSING_SCORE

 49class RecordBook:
 50    """
 51    Зачетная книжка конкретного студента, хранящая историю всех оценок.
 52
 53    Attributes:
 54        student_id (str): Идентификатор владельца (Student.person_id).
 55        book_number (str): Уникальный номер бланка зачетной книжки.
 56    """
 57
 58    def __init__(self, student_id: str, book_number: str) -> None:
 59        self.student_id = student_id
 60        self.book_number = book_number
 61        self._grades: list[Grade] = []
 62
 63    def add_grade(self, grade: Grade) -> None:
 64        """
 65        Вносит новую оценку в зачетную книжку.
 66
 67        Args:
 68            grade (Grade): Объект выставляемой оценки.
 69
 70        Raises:
 71            ValueError: Если экзаменационная оценка по этому предмету уже стоит.
 72        """
 73        if grade.is_exam and any(g.subject_name == grade.subject_name and g.is_exam for g in self._grades):
 74            raise ValueError(f"Экзаменационная оценка по {grade.subject_name} уже стоит в зачетке {self.book_number}")
 75        self._grades.append(grade)
 76
 77    @property
 78    def average_score(self) -> float:
 79        """Агрегация: вычисляет средний балл по всем записям."""
 80        if not self._grades:
 81            return 0.0
 82        total = sum(g.score for g in self._grades)
 83        return round(total / len(self._grades), 2)
 84
 85    def get_debts(self) -> list[str]:
 86        """
 87        Анализ успеваемости: формирует список предметов с академической задолженностью.
 88
 89        Returns:
 90            list[str]: Названия несданных дисциплин.
 91        """
 92        return [grade.subject_name for grade in self._grades if not grade.is_passed]
 93
 94    def clear_debts_for_subject(self, subject_name: str) -> None:
 95        """
 96        Удаляет неудовлетворительные оценки по предмету после успешной пересдачи.
 97
 98        Args:
 99            subject_name (str): Название дисциплины, по которой закрыт долг.
100        """
101        self._grades = [g for g in self._grades if not (g.subject_name == subject_name and not g.is_passed)]
102
103    @property
104    def is_excellent(self) -> bool:
105        """Бизнес-правило: является ли студент круглым отличником (все оценки >= 9)."""
106        if not self._grades:
107            return False
108        return all(g.score >= 9 for g in self._grades)
109
110    def get_best_subject(self) -> str | None:
111        """
112        Ищет предмет с максимальным баллом.
113
114        Returns:
115            str | None: Название предмета или None, если оценок нет.
116        """
117        if not self._grades:
118            return None
119        best_grade = max(self._grades, key=lambda g: g.score)
120        return best_grade.subject_name

123class AcademicStatement:
124    """
125    Экзаменационная ведомость группы для конкретной дисциплины.
126
127    Attributes:
128        statement_id (str): Уникальный номер ведомости.
129        subject (Subject): Учебная дисциплина.
130        group (AcademicGroup): Экзаменуемая группа.
131        examiner (Lecturer): Преподаватель, принимающий экзамен.
132        is_closed (bool): Статус документа (закрыт/открыт).
133    """
134
135    def __init__(self, subject: Subject, group: AcademicGroup, examiner: Lecturer) -> None:
136        self.statement_id = str(uuid.uuid4())
137        self.subject = subject
138        self.group = group
139        self.examiner = examiner
140        self.is_closed = False
141        self._results: dict[str, Grade] = {}  # Ключ: student_id, Значение: Grade
142
143    def add_result(self, student: Student, score: int) -> None:
144        """
145        Вносит оценку конкретного студента в ведомость.
146
147        Args:
148            student (Student): Сдающий студент.
149            score (int): Выставленный балл.
150
151        Raises:
152            ValueError: Если ведомость уже закрыта.
153        """
154        if self.is_closed:
155            raise ValueError("Ведомость уже закрыта и передана в деканат.")
156
157        grade = Grade(self.subject.name, score, is_exam=True)
158        self._results[student.person_id] = grade
159
160    def close_statement(self) -> None:
161        """Блокирует ведомость от дальнейших изменений."""
162        self.is_closed = True
163
164    @property
165    def pass_rate(self) -> float:
166        """Агрегация: вычисляет процент успешной сдачи (успеваемость группы)."""
167        if not self._results:
168            return 0.0
169        passed = sum(1 for grade in self._results.values() if grade.is_passed)
170        return round((passed / len(self._results)) * 100, 2)

176

178        sheet_id (str): Уникальный номер бланка.
179        student (Student): Направляемый студент.
180        subject (Subject): Дисциплина для пересдачи.
181        max_attempts (int): Максимально разрешенное количество попыток.
182        attempts_used (int): Использованное количество попыток.
183        is_valid (bool): Действителен ли бегунок. False - 
184    """
185    DEFAULT_MAX_ATTEMPTS = 3
186
187    def __init__(self, student: Student, subject: Subject, max_attempts: int = DEFAULT_MAX_ATTEMPTS) -> None:
188        self.sheet_id = str(uuid.uuid4())
189        self.student = student
190        self.subject = subject
191        self.max_attempts = max_attempts
192        self.attempts_used = 0
193        self.is_valid = True
194
195    def register_attempt(self, score: int) -> Grade:
196        """
197        Фиксирует попытку сдать экзамен по бегунку.
198
199        Args:
200            score (int): Полученный балл.
201
202        Returns:
203            Grade: Объект итоговой оценки.
204
205        Raises:
206            ValueError: Если лимит попыток исчерпан или бегунок недействителен.
207        """
208        if not self.is_valid:
209            raise ValueError("Бегунок просрочен или лимит попыток исчерпан.")
210
211        self.attempts_used += 1
212
213        if self.attempts_used >= self.max_attempts:
214            self.is_valid = False
215
216        return Grade(self.subject.name, score, is_exam=True)
