 46class Student(Person):
 47    """
 48    Студент университета.
 49
 50    Attributes:
 51        record_book_number (str): Номер зачетной книжки.
 52        is_active (bool): Статус обучения (False, если отчислен или в академе).
 53        has_scholarship (bool): Наличие академической стипендии.
 54        financial_debt (float): Текущий долг за платное обучение или общежитие.
 55    """
 56
 57    def __init__(self, first_name: str, last_name: str, middle_name: str = "", record_book_number: str = "") -> None:
 58        super().__init__(first_name, last_name, middle_name)
 59        self.record_book_number = record_book_number
 60        self.is_active = True
 61        self.has_scholarship = False
 62        self.financial_debt = 0.0
 63
 64    def expel(self) -> None:
 65        """Отчисляет студента, деактивируя его статус и лишая стипендии."""
 66        self.is_active = False
 67        self.has_scholarship = False
 68
 69    def grant_scholarship(self) -> None:
 70        """
 71        Назначает академическую стипендию.
 72
 73        Raises:
 74            ValueError: Если попытка назначить стипендию неактивному (отчисленному) студенту.
 75        """
 76        if not self.is_active:
 77            raise ValueError(f"Нельзя назначить стипендию отчисленному студенту {self.full_name}")
 78        self.has_scholarship = True
 79
 80    @property
 81    def is_debtor(self) -> bool:
 82        """Проверяет, имеет ли студент непогашенную финансовую задолженность."""
 83        return self.financial_debt > 0.0
 84
 85    def pay_debt(self, amount: float) -> None:
 86        """
 87        Частично или полностью погашает финансовую задолженность студента.
 88
 89        Args:
 90            amount (float): Сумма внесенного платежа.
 91
 92        Raises:
 93            ValueError: Если сумма платежа отрицательная или равна нулю.
 94        """
 95        if amount <= 0:
 96            raise ValueError("Сумма оплаты должна быть положительной.")
 97        if amount > self.financial_debt:
 98            self.financial_debt = 0.0
 99        else:
100            self.financial_debt -= amount

145    """

148    Attributes:
149        degree (str): Ученая степень.

152    DEFAULT_POSITION = "Преподаватель"
153
154    def __init__(self, first_name: str, last_name: str, middle_name: str = "", degree: str = DEFAULT_DEGREE) -> None:
155        super().__init__(first_name, last_name, middle_name, position=self.DEFAULT_POSITION)
156        self.degree = degree
157        self._subjects: list[str] = []
158
159    @property
160    def subjects_count(self) -> int:
161        """Агрегация: количество читаемых преподавателем дисциплин."""
162        return len(self._subjects)
163
164    def assign_subject(self, subject_name: str) -> None:
165        """
166        Назначает преподавателю новую дисциплину для ведения.

168
169        Args:
170            subject_name (str): Название учебной дисциплины.
171        """
172        if subject_name not in self._subjects:
173            self._subjects.append(subject_name)
174
175    def update_degree(self, new_degree: str) -> None:
176        """
177        Обновляет ученую степень преподавателя (например, после защиты диссертации).
178
179        Raises:
180            ValueError: Если передана пустая строка.
181        """
182        if not new_degree.strip():
183            raise ValueError("Ученая степень не может быть пустой строкой")
184        self.degree = new_degree
185
186    def can_teach(self, subject_name: str) -> bool:
187        """
188        Проверяет компетенцию: ведет ли преподаватель указанную дисциплину.
189        """
190        return subject_name in self._subjects
