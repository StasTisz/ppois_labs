lab3.domain.operations

Модуль операционной деятельности аэропорта.

Отвечает за логику управления рейсами, билетами, багажом и расписанием. Связывает воедино инфраструктуру (гейты, карусели), флот (самолеты) и людей (пассажиров). Обеспечивает транзакционность при регистрации на рейс и посадке.

 28class Baggage:
 29    """
 30    Единица багажа пассажира.
 31
 32    Attributes:
 33        baggage_id (str): Уникальный RFID-номер багажной бирки.
 34        weight_kg (float): Фактический вес чемодана в килограммах.
 35        owner_id (str): Идентификатор пассажира-владельца.
 36    """
 37    MAX_STANDARD_WEIGHT = 23.0
 38
 39    def __init__(self, weight_kg: float, owner_id: str) -> None:
 40        self.baggage_id = str(uuid.uuid4())
 41        self.weight_kg = weight_kg
 42        self.owner_id = owner_id
 43
 44    def check_weight(self) -> None:
 45        """
 46        Проверяет багаж на соответствие нормам провоза.
 47
 48        Raises:
 49            BaggageOverweightException: Если вес превышает допустимый лимит.
 50        """
 51        if self.weight_kg > self.MAX_STANDARD_WEIGHT:
 52            raise BaggageOverweightException(
 53                f"Перевес: {self.weight_kg} кг. Максимум: {self.MAX_STANDARD_WEIGHT} кг."
 54            )

 57class Ticket:
 58    """
 59    Маршрутная квитанция (электронный билет) пассажира.
 60
 61    Attributes:
 62        ticket_number (str): Уникальный 8-значный буквенно-цифровой номер билета.
 63        passenger_id (str): Идентификатор владельца (связь с Person.person_id).
 64        flight_number (str): Номер рейса (например, "B2-973").
 65        seat_class (str): Класс обслуживания (Economy, Business, First).
 66        price (float): Стоимость приобретенного билета.
 67        is_used (bool): Флаг гашения билета после успешной регистрации.
 68    """
 69
 70    def __init__(
 71            self, passenger_id: str, flight_number: str, seat_class: str, price: float
 72    ) -> None:
 73        self.ticket_number = str(uuid.uuid4())[:8].upper()
 74        self.passenger_id = passenger_id
 75        self.flight_number = flight_number
 76        self.seat_class = seat_class
 77        self.price = price
 78        self.is_used = False
 79
 80    def use_ticket(self) -> None:
 81        """
 82        Помечает билет как использованный (гасит его).
 83
 84        Raises:
 85            InvalidTicketException: Если билет уже был погашен ранее.
 86        """
 87        if self.is_used:
 88            raise InvalidTicketException(f"Билет {self.ticket_number} уже был использован.")
 89        self.is_used = True

152class Flight(ABC):
153    """
154    Базовый абстрактный класс авиарейса.
155
156    Attributes:
157        flight_number (str): Уникальный номер рейса.
158        airline (Airline): Компания-оператор.
159        aircraft (Aircraft): Воздушное судно, выполняющее рейс.
160        plan (FlightPlan): План полета.
161        status (str): Текущий статус рейса (Scheduled, Boarding, InFlight, Landed, Delayed, Cancelled).
162        delay_reason (str | None): Причина задержки рейса (если есть).
163    """
164
165    def __init__(
166            self, flight_number: str, airline: Airline, aircraft: Aircraft, plan: FlightPlan
167    ) -> None:
168        self.flight_id = str(uuid.uuid4())
169        self.flight_number = flight_number
170        self.airline = airline
171        self.aircraft = aircraft
172        self.plan = plan
173        self.status = "Scheduled"
174        self.delay_reason: str | None = None
175
176    def delay(self, reason: str) -> None:
177        """
178        Переводит рейс в статус 'Delayed' с указанием причины.
179
180        Raises:
181            ValueError: Если рейс уже отменен, находится в воздухе или приземлился.
182        """
183        if self.status in ["InFlight", "Landed", "Cancelled"]:
184            raise ValueError(f"Невозможно отложить рейс в статусе {self.status}.")
185        self.status = "Delayed"
186        self.delay_reason = reason
187
188    def cancel(self) -> None:
189        """Полностью отменяет рейс."""
190        self.status = "Cancelled"

193class DepartureFlight(Flight):
194    """
195    Вылетающий рейс. Управляет процессом регистрации, багажом и посадкой.
196    """
197
198    def __init__(
199            self, flight_number: str, airline: Airline, aircraft: Aircraft, plan: FlightPlan
200    ) -> None:
201        super().__init__(flight_number, airline, aircraft, plan)
202        self.assigned_gate: Gate | None = None
203        self._manifest: list[Passenger] = []
204        self._cargo_hold: list[Baggage] = []
205        self._seat_counter = 0
206
207    @property
208    def manifest(self) -> list[Passenger]:
209        """Безопасная копия списка зарегистрированных пассажиров."""
210        return self._manifest.copy()
211
212    @property
213    def cargo_hold(self) -> list[Baggage]:
214        """Безопасная копия списка загруженного багажа."""
215        return self._cargo_hold.copy()
216
217    def assign_gate(self, gate: Gate) -> None:
218        """Привязывает инфраструктурный гейт к текущему рейсу."""
219        self.assigned_gate = gate
220        gate.assign_flight(self)
221
222    def check_in_passenger(self, passenger: Passenger, ticket: Ticket) -> BoardingPass:
223        """
224        Регистрирует пассажира на рейс, гасит билет и генерирует посадочный талон.
225
226        Raises:
227            FlightDelayedException: Если регистрация закрыта или рейс отложен/отменен.
228            InvalidTicketException: Если билет не принадлежит этому рейсу.
229        """
230        if self.status in ["Delayed", "Cancelled"]:
231            raise FlightDelayedException(f"Регистрация невозможна: рейс {self.status}.")
232        if ticket.flight_number != self.flight_number:
233            raise InvalidTicketException("Билет оформлен на другой рейс.")
234
235        ticket.use_ticket()
236        self._manifest.append(passenger)
237
238        # Алгоритм выдачи места (ряд + буква)
239        row = (self._seat_counter // 6) + 1
240        letter = chr(65 + (self._seat_counter % 6))
241        seat_number = f"{row}{letter}"
242        self._seat_counter += 1
243
244        gate_number = self.assigned_gate.number if self.assigned_gate else "TBD"
245        return BoardingPass(ticket, seat_number, gate_number)
246
247    def load_baggage(self, baggage: Baggage) -> None:
248        """Грузит проверенный багаж в трюм самолета."""
249        baggage.check_weight()
250        self._cargo_hold.append(baggage)
251
252    def start_boarding(self) -> None:
253        """Объявляет посадку и открывает привязанный гейт."""
254        if not self.assigned_gate:
255            raise ValueError("Гейт не назначен.")
256        self.status = "Boarding"
257        self.assigned_gate.open_gate()
258

260class ArrivalFlight(Flight):
261    """
262    Прибывающий рейс. Взаимодействует с каруселями для выдачи багажа.
263    """
264
265    def __init__(
266            self, flight_number: str, airline: Airline, aircraft: Aircraft, plan: FlightPlan
267    ) -> None:
268        super().__init__(flight_number, airline, aircraft, plan)
269        self.assigned_carousel: BaggageCarousel | None = None
270        self.status = "InFlight"
271
272    def assign_carousel(self, carousel: BaggageCarousel) -> None:
273        """Назначает карусель для выгрузки багажа пассажирам."""
274        self.assigned_carousel = carousel
275
276    def land(self) -> None:
277        """Регистрирует приземление, меняет статус борта и запускает ленту багажа."""
278        self.status = "Landed"
279        self.aircraft.land()
280        if self.assigned_carousel:
281            self.assigned_carousel.activate(self)

284class Schedule:
285    """
286    Электронное табло расписания рейсов (реестр полетов).
287    """
288
289    def __init__(self) -> None:
290        self._flights: list[Flight] = []
291
292    def add_flight(self, flight: Flight) -> None:
293        """Добавляет рейс в общее расписание."""
294        self._flights.append(flight)
295
296    def get_departures(self) -> list[DepartureFlight]:
297        """Возвращает список всех вылетающих рейсов."""
298        return [f for f in self._flights if isinstance(f, DepartureFlight)]
299
300    def get_arrivals(self) -> list[ArrivalFlight]:
301        """Возвращает список всех прибывающих рейсов."""
302        return [f for f in self._flights if isinstance(f, ArrivalFlight)]
