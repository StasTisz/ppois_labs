lab3.domain.infrastructure

Модуль инфраструктуры аэропорта.

Содержит классы, описывающие физические и логические объекты на территории аэропорта. Служит основой для взаимодействия флота и пассажиров.

 26class Runway:
 27    """
 28    Взлетно-посадочная полоса.
 29    Отвечает за координацию занятости при взлетах и посадках.
 30
 31    Attributes:
 32        runway_id (str): Внутренний уникальный идентификатор.
 33        number (str): Навигационное обозначение полосы (например, "27R").
 34        length (int): Длина полосы в метрах (влияет на типы принимаемых судов).
 35        is_busy (bool): Флаг текущей занятости маневрирующим бортом.
 36        current_aircraft (Aircraft | None): Борт, находящийся на полосе.
 37    """
 38
 39    def __init__(self, number: str, length: int) -> None:
 40        self.runway_id = str(uuid.uuid4())
 41        self.number = number
 42        self.length = length
 43        self.is_busy = False
 44        self.current_aircraft: Aircraft | None = None
 45
 46    def occupy(self, aircraft: Aircraft) -> None:
 47        """
 48        Занимает полосу указанным воздушным судном.
 49
 50        Raises:
 51            RunwayBusyException: Если полоса уже занята другим бортом.
 52        """
 53        if self.is_busy:
 54            raise RunwayBusyException(f"Полоса {self.number} уже занята.")
 55        self.is_busy = True
 56        self.current_aircraft = aircraft
 57
 58    def clear(self) -> None:
 59        """Освобождает полосу после завершения взлета или руления на перрон."""
 60        self.is_busy = False
 61        self.current_aircraft = None

 64class Gate:
 65    """
 66    Выход на посадку (гейт) в терминале.
 67    Связывает инфраструктуру терминала с конкретным авиарейсом.
 68
 69    Attributes:
 70        gate_id (str): Внутренний уникальный идентификатор.
 71        number (str): Номер гейта для табло (например, "A12").
 72        is_open (bool): Открыта ли физически посадка пассажиров.
 73        current_flight (Flight | None): Рейс, обслуживаемый у данного гейта.
 74    """
 75
 76    def __init__(self, number: str) -> None:
 77        self.gate_id = str(uuid.uuid4())
 78        self.number = number
 79        self.is_open = False
 80        self.current_flight: Flight | None = None
 81
 82    def assign_flight(self, flight: Flight) -> None:
 83        """
 84        Привязывает рейс к гейту для подготовки к посадке.
 85
 86        Raises:
 87            ValueError: Если гейт не освобожден от предыдущего рейса.
 88        """
 89        if self.current_flight is not None:
 90            raise ValueError(f"Гейт {self.number} уже обслуживает другой рейс.")
 91        self.current_flight = flight
 92
 93    def open_gate(self) -> None:
 94        """Открывает двери гейта для посадки по талонам."""
 95        if self.current_flight is None:
 96            raise GateNotAssignedException(f"Нельзя открыть гейт {self.number}: рейс не назначен.")
 97        self.is_open = True
 98
 99    def close_gate(self) -> None:
100        """Останавливает пропуск пассажиров на рейс."""
101        self.is_open = False
102
103    def clear_gate(self) -> None:
104        """Полностью освобождает гейт после отбытия судна."""
105        self.close_gate()
106        self.current_flight = None

109class CheckInCounter:
110    """
111    Стойка регистрации пассажиров и приема багажа.
112
113    Attributes:
114        counter_id (str): Внутренний уникальный идентификатор.
115        number (int): Номер стойки регистрации.
116        is_open (bool): Статус работы.
117        current_agent (CheckInAgent | None): Сотрудник, авторизованный за стойкой.
118        assigned_flight (Flight | None): Рейс, на который ведется прием.
119    """
120
121    def __init__(self, number: int) -> None:
122        self.counter_id = str(uuid.uuid4())
123        self.number = number
124        self.is_open = False
125        self.current_agent: CheckInAgent | None = None
126        self.assigned_flight: Flight | None = None
127        self._accepted_baggage: list[Baggage] = []
128
129    def open_counter(self, agent: CheckInAgent, flight: Flight) -> None:
130        """Открывает смену на стойке регистрации."""
131        self.current_agent = agent
132        self.assigned_flight = flight
133        self.is_open = True
134
135    def close_counter(self) -> None:
136        """Закрывает стойку и отправляет принятый багаж в систему сортировки."""
137        self.is_open = False
138        self.current_agent = None
139        self.assigned_flight = None
140        self._accepted_baggage.clear()
141
142    def accept_baggage(self, baggage: Baggage) -> None:
143        """Принимает единицу багажа на конвейерную ленту."""
144        if not self.is_open:
145            raise ValueError("Стойка регистрации закрыта.")
146        self._accepted_baggage.append(baggage)

171class Lounge:
172    """
173    Зал ожидания для пассажиров (бизнес-залы, VIP-комнаты).
174    """
175
176    def __init__(self, name: str, capacity: int, is_vip: bool = False) -> None:
177        self.lounge_id = str(uuid.uuid4())
178        self.name = name
179        self.capacity = capacity
180        self.is_vip = is_vip
181        self._passengers: list[Passenger] = []
182
183    def enter(self, passenger: Passenger) -> None:
184        """Впускает пассажира с проверкой лимита вместимости."""
185        if len(self._passengers) >= self.capacity:
186            raise CapacityExceededException(f"Зал {self.name} переполнен.")
187        self._passengers.append(passenger)
188
189    def exit(self, passenger: Passenger) -> None:
190        """Регистрирует выход пассажира из зала."""
191        if passenger in self._passengers:
192            self._passengers.remove(passenger)
193
194    @property
195    def current_occupancy(self) -> int:
196        return len(self._passengers)

232class ControlTower:
233    """
234    Диспетчерская вышка. Центральный орган управления воздушным движением.
235    Отвечает за выдачу разрешений на взлет/посадку с учетом метеоусловий.
236    """
237
238    def __init__(self) -> None:
239        self.tower_id = str(uuid.uuid4())
240        self._runways: list[Runway] = []
241        self.weather_is_safe = True
242
243    def add_runway(self, runway: Runway) -> None:
244        """Берет ВПП под управление вышки."""
245        self._runways.append(runway)
246
247    def update_weather(self, is_safe: bool) -> None:
248        """Обновляет статус безопасности метеоусловий."""
249        self.weather_is_safe = is_safe
250
251    def request_takeoff(self, aircraft: Aircraft) -> Runway:
252        """
253        Запрашивает коридор для взлета.
254
255        Raises:
256            WeatherWarningException: При штормовом предупреждении.
257            RunwayBusyException: Если нет свободных полос.
258        """
259        if not self.weather_is_safe:
260            raise WeatherWarningException("Взлет запрещен из-за сложных метеоусловий.")
261
262        for runway in self._runways:
263            if not runway.is_busy:
264                runway.occupy(aircraft)
265                return runway
266
267        raise RunwayBusyException("Все ВПП заняты, ожидайте слота.")
268
269    def request_landing(self, aircraft: Aircraft) -> Runway:
270        """
271        Запрашивает коридор для посадки.
272        """
273        if not self.weather_is_safe:
274            raise WeatherWarningException("Посадка запрещена (аэропорт закрыт по метеоусловиям).")
275
276        for runway in self._runways:
277            if not runway.is_busy:
278                runway.occupy(aircraft)
279                return runway
280
281        raise RunwayBusyException("Борт отправлен в зону ожидания: все ВПП заняты.")

284class Hangar:
285    """
286    Технический ангар для длительной стоянки и ремонта воздушных судов.
287    """
288
289    def __init__(self, number: str, capacity: int) -> None:
290        self.hangar_id = str(uuid.uuid4())
291        self.number = number
292        self.capacity = capacity
293        self._aircrafts: list[Aircraft] = []
294
295    def store_aircraft(self, aircraft: Aircraft) -> None:
296        """Отправляет борт в ангар с проверкой наличия мест."""
297        if len(self._aircrafts) >= self.capacity:
298            raise CapacityExceededException(f"Ангар {self.number} заполнен.")
299        self._aircrafts.append(aircraft)
300
301    def release_aircraft(self, aircraft: Aircraft) -> None:
302        """Выпускает борт из ангара на перрон."""
303        if aircraft in self._aircrafts:
304            self._aircrafts.remove(aircraft)
