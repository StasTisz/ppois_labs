lab3.domain.fleet

Модуль транспортного парка (флота) аэропорта.

Содержит иерархию классов для воздушных судов (пассажирских, грузовых, частных) и наземной спецтехники (тягачи, заправщики, автобусы). Обеспечивает полиморфизм в процедурах технического обслуживания и инкапсуляцию внутренних состояний (топливо, загрузка, пассажиры).

 17class Vehicle(ABC):
 18    """
 19    Базовый абстрактный класс для любого транспортного средства в аэропорту.
 20    Определяет общий контракт технического обслуживания.
 21
 22    Attributes:
 23        vehicle_id (str): Внутренний уникальный идентификатор транспорта.
 24        model (str): Модель транспортного средства (например, "Boeing 737").
 25        max_speed (float): Максимальная скорость в км/ч.
 26        requires_maintenance (bool): Флаг необходимости проведения ТО.
 27        operating_hours (float): Количество часов наработки после последнего ТО.
 28    """
 29
 30    def __init__(self, model: str, max_speed: float) -> None:
 31        self.vehicle_id = str(uuid.uuid4())
 32        self.model = model
 33        self.max_speed = max_speed
 34        self.requires_maintenance = False
 35        self.operating_hours = 0.0
 36
 37    @abstractmethod
 38    def perform_maintenance(self) -> None:
 39        """
 40        Полиморфный контракт проведения ТО.
 41        Каждый конкретный тип техники реализует свои уникальные процедуры.
 42        """
 43        pass

 46class Aircraft(Vehicle, ABC):
 47    """
 48    Абстрактный класс воздушного судна.
 49    Отвечает за базовые механики полета и заправки.
 50
 51    Attributes:
 52        tail_number (str): Бортовой регистрационный номер (например, "RA-12345").
 53        fuel_capacity (float): Максимальный объем топливных баков в литрах.
 54        is_in_flight (bool): Находится ли судно в данный момент в воздухе.
 55    """
 56
 57    def __init__(self, model: str, max_speed: float, tail_number: str, fuel_capacity: float) -> None:
 58        super().__init__(model, max_speed)
 59        self.tail_number = tail_number
 60        self.fuel_capacity = fuel_capacity
 61        self._current_fuel = 0.0  # Защищенное поле, меняется только через refuel()
 62        self.is_in_flight = False
 63
 64    @property
 65    def current_fuel(self) -> float:
 66        """Текущий уровень топлива в баках (доступно только для чтения)."""
 67        return self._current_fuel
 68
 69    def refuel(self, amount: float) -> None:
 70        """
 71        Заправляет самолет указанным объемом топлива.
 72
 73        Raises:
 74            ValueError: При попытке залить отрицательный объем.
 75            CapacityExceededException: При превышении емкости баков.
 76        """
 77        if amount <= 0:
 78            raise ValueError("Объем заправки должен быть положительным.")
 79        if self._current_fuel + amount > self.fuel_capacity:
 80            raise CapacityExceededException(f"Баки переполнены. Максимум: {self.fuel_capacity} л.")
 81        self._current_fuel += amount
 82
 83    def take_off(self) -> None:
 84        """Переводит борт в статус полета."""
 85        self.is_in_flight = True
 86
 87    def land(self) -> None:
 88        """
 89        Регистрирует приземление борта.
 90        После каждого рейса автоматически выставляется флаг необходимости ТО.
 91        """
 92        self.is_in_flight = False
 93        self.requires_maintenance = True

128    def perform_maintenance(self) -> None:

130        self.oxygen_masks_tested = True
131        self.cabin_pressurization_ok = True

216        """Отправляет технику на задание, блокируя её статус."""

219        self.is_available = False
220
221    def release(self) -> None:
222        """Освобождает технику после выполнения задачи."""
223        self.is_available = True

259class FuelTruck(GroundVehicle):
260    """
261    Автомобиль-топливозаправщик. Перекачивает топливо в баки судов.
262    """
263
264    def __init__(self, model: str, max_speed: float, license_plate: str, tank_capacity: float) -> None:
265        super().__init__(model, max_speed, license_plate)
266        self.tank_capacity = tank_capacity
267        self._current_fuel = tank_capacity  # Выезжает с полным баком
268        self.pump_calibrated = False
269        self.filter_replaced = False
270
271    @property
272    def current_fuel(self) -> float:
273        """Текущий остаток топлива в цистерне заправщика."""
274        return self._current_fuel
275
276    def refuel_aircraft(self, aircraft: Aircraft, amount: float) -> None:
277        """
278        Перекачивает керосин из цистерны грузовика в баки самолета.
279
280        Args:
281            aircraft (Aircraft): Борт, который необходимо заправить.
282            amount (float): Объем перекачиваемого топлива.
283        """
284        if self._current_fuel < amount:
285            raise CapacityExceededException("В цистерне топливозаправщика недостаточно керосина.")
286
287        self.dispatch()
288        aircraft.refuel(amount)
289        self._current_fuel -= amount
290        self.release()
291
292    def perform_maintenance(self) -> None:
293        """ТО заправщика: регламентная замена фильтров и калибровка счетчика."""
294        self.pump_calibrated = True
295        self.filter_replaced = True
296        self.requires_maintenance = False
297        self.operating_hours = 0.0
